Create the `BookingStatusTabs` Client Component inline-but-separate pattern: since the page is `async` (Server Component), the tabs UI must be a separate named export defined in a client-boundary file. Use this approach:

Define `BookingStatusTabs` at the top of the file WITHOUT `"use client"` (make it purely presentational using only `<Link>` from next/link, no hooks) — `<Link>` from next/link works in Server Components too. This avoids the need for a client component entirely.

```typescript
// app/[locale]/admin/bookings/page.tsx
import Link from "next/link";
import { getAdminBookings } from "@/lib/queries/admin";
import { getDictionary } from "@/lib/i18n/get-dictionary";
import { isLocale } from "@/lib/i18n/config";
import { localizedPath } from "@/lib/i18n/path";
import { Table } from "@/components/ui/table";
import { Badge } from "@/components/ui/badge";
import { PageHeader } from "@/components/ui/page-header";
import { EmptyState } from "@/components/ui/empty-state";

type Props = {
  params: Promise<{ locale: string }>;
  searchParams: Promise<{ status?: string }>;
};

const VALID_TABS = ["all", "pending", "confirmed", "completed", "cancelled"] as const;
type TabValue = (typeof VALID_TABS)[number];

// Purely presentational — no hooks, so no "use client" needed
function BookingStatusTabs({
  current,
  basePath,
  labels,
}: {
  current: string;
  basePath: string;
  labels: Record<TabValue, string>;
}) {
  return (
    <div className="mb-4 flex gap-1 border-b border-academy-line">
      {VALID_TABS.map((tab) => (
        <Link
          key={tab}
          href={tab === "all" ? basePath : `${basePath}?status=${tab}`}
          className={[
            "px-4 py-2 text-sm font-medium transition-colors",
            current === tab
              ? "border-b-2 border-academy-navy text-academy-navy"
              : "text-academy-slate hover:text-academy-navy",
          ].join(" ")}
        >
          {labels[tab]}
        </Link>
      ))}
    </div>
  );
}

export default async function AdminBookingsPage({ params, searchParams }: Props) {
  const { locale } = await params;
  const { status } = await searchParams;  // Next.js 16: MUST await searchParams
  const dict = getDictionary(isLocale(locale) ? locale : "de");
  const t = dict.admin.bookings;

  const currentTab: TabValue =
    VALID_TABS.includes(status as TabValue) && status !== "all"
      ? (status as TabValue)
      : "all";

  const bookings = await getAdminBookings(currentTab === "all" ? undefined : currentTab);
  const basePath = localizedPath(isLocale(locale) ? locale : "de", "/admin/bookings");

  const tabLabels: Record<TabValue, string> = {
    all: t.tabs.all,
    pending: t.tabs.pending,
    confirmed: t.tabs.confirmed,
    completed: t.tabs.completed,
    cancelled: t.tabs.cancelled,
  };

  const columns = [
    {
      key: "student",
      header: t.colStudent,
      render: (row: (typeof bookings)[0]) =>
        row.students?.profiles?.full_name ?? "—",
    },
    {
      key: "teacher",
      header: t.colTeacher,
      render: (row: (typeof bookings)[0]) =>
        row.teachers?.profiles?.full_name ?? "—",
    },
    {
      key: "date",
      header: t.colDate,
      render: (row: (typeof bookings)[0]) =>
        new Date(row.start_time).toLocaleDateString("de-CH"),
    },
    {
      key: "credits",
      header: t.colCredits,
      render: (row: (typeof bookings)[0]) => row.credits_reserved,
    },
    {
      key: "status",
      header: t.colStatus,
      render: (row: (typeof bookings)[0]) => (
        <Badge variant="muted">{row.status}</Badge>
      ),
    },
  ];

  return (
    <div>
      <PageHeader title={t.title} />
      <BookingStatusTabs current={currentTab} basePath={basePath} labels={tabLabels} />
      <Table
        columns={columns}
        rows={bookings}
        emptyState={<EmptyState icon="calendar" title={t.empty} description="" />}
      />
    </div>
  );
}
```
