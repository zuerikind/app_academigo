```typescript
const t = dict.admin.students;
const columns = [
  {
    key: "name",
    header: t.colName,
    render: (row) => row.profiles?.full_name ?? "—",
  },
  {
    key: "email",
    header: t.colEmail,
    render: (row) => row.profiles?.email ?? "—",
  },
  {
    key: "credits",
    header: t.colCredits,
    render: (row) => {
      const sc = row.student_credits;
      if (!sc) return "—";
      const available = sc.total_credits - sc.used_credits - sc.reserved_credits;
      return `${available} / ${sc.total_credits}`;
    },
  },
  {
    key: "joined",
    header: t.colJoined,
    render: (row) => new Date(row.created_at).toLocaleDateString("de-CH"),
  },
];
```
