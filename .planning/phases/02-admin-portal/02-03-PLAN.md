The layout does two things:

2. Wraps children with `DashboardLayout` using `getAdminNav(dict, locale)` so all admin pages get the sidebar/nav without repeating it

Look at `app/[locale]/teacher/layout.tsx` for the `requireRoleFromParams` pattern. Look at existing teacher/student dashboard pages for how `DashboardLayout` is invoked and what props it expects (navItems, children, locale, dict or similar — read the component's props before writing).

```typescript
// app/[locale]/admin/layout.tsx
import { requireRoleFromParams } from "@/lib/auth/session";
import { getDictionary } from "@/lib/i18n/get-dictionary";
import { isLocale } from "@/lib/i18n/config";
import { getAdminNav } from "@/config/navigation";
import { DashboardLayout } from "@/components/layout/dashboard-layout";

export default async function AdminLayout({
  children,
  params,
}: {
  children: React.ReactNode;
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;
  await requireRoleFromParams("admin", locale);

  const dict = getDictionary(isLocale(locale) ? locale : "de");
  const navItems = getAdminNav(dict, isLocale(locale) ? locale : "de");

  return (
    <DashboardLayout navItems={navItems} locale={isLocale(locale) ? locale : "de"} dict={dict}>
      {children}
    </DashboardLayout>
  );
}
```

IMPORTANT: Read `components/layout/dashboard-layout.tsx` before writing to confirm the exact prop names and types DashboardLayout expects. Adjust the JSX above to match exactly — do not guess prop names.
