const WEEKDAY_LABELS: Record<string, string[]> = {

};
function getWeekdayLabel(weekday: number, locale: string): string {
  const labels = WEEKDAY_LABELS[locale] ?? WEEKDAY_LABELS["en"];
  return labels[weekday] ?? String(weekday);
}
