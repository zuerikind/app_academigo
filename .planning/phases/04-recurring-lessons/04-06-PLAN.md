```typescript
import { type NextRequest } from "next/server";
import { createServiceClient } from "@/lib/supabase/service";
import { computeOccurrences } from "@/lib/utils/schedule";

export async function GET(request: NextRequest): Promise<Response> {
  const authHeader = request.headers.get("authorization");
  const cronSecret = process.env.CRON_SECRET;
  if (!cronSecret || authHeader !== `Bearer ${cronSecret}`) {
    return new Response("Unauthorized", { status: 401 });
  }

  // MUST use service client — no user session in cron context (Pitfall 2)
  const supabase = createServiceClient();

  const { data: schedules, error: scheduleError } = await supabase
    .from("recurring_schedules")
    .select("id, student_id, teacher_id, weekday, start_time, end_time, status")
    .eq("status", "active");  // SCHED-03: only active schedules

  if (scheduleError) {
    return Response.json({ ok: false, error: scheduleError.message }, { status: 500 });
  }

  const now = new Date();
  const windowEnd = new Date(now.getTime() + 42 * 24 * 60 * 60 * 1000); // 6 weeks = 42 days
  let generated = 0;

  for (const schedule of schedules ?? []) {
    // Double-guard: check status again inside loop (Pitfall 5 race condition mitigation)
    if (schedule.status !== "active") continue;

    const occurrences = computeOccurrences(schedule, now, windowEnd);

    for (const occurrence of occurrences) {
      const { error } = await supabase
        .from("lessons")
        .insert(occurrence);
      // ON CONFLICT DO NOTHING is enforced by the partial UNIQUE INDEX (schedule_id, start_time)
      // A conflict error from the unique constraint means duplicate — skip silently
      if (!error) {
        generated++;
      }
      // If error is a unique violation (code 23505), that's fine — idempotent
      // Other errors are logged but don't abort the whole run
    }
  }

  return Response.json({ ok: true, generated });
}
```

**Note on INSERT idempotency:** The UNIQUE partial index `lessons_schedule_start_unique ON (schedule_id, start_time) WHERE schedule_id IS NOT NULL` (created in 04-01 migration) handles the ON CONFLICT. Supabase returns an error for constraint violations — the code checks `if (!error)` to count only new insertions. Alternatively, use `.upsert(occurrence, { onConflict: 'schedule_id,start_time', ignoreDuplicates: true })` if the Supabase client version supports it — check the existing codebase first.
