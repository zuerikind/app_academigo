```sql
CREATE OR REPLACE FUNCTION grant_credits(p_student_id UUID, p_credits INTEGER)
RETURNS void AS $$
BEGIN
  UPDATE student_credits SET total_credits = total_credits + p_credits, updated_at = now()
  WHERE student_id = p_student_id;
  IF NOT FOUND THEN RAISE EXCEPTION 'student_credits_not_found'; END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER SET search_path = public;

CREATE OR REPLACE FUNCTION grant_subscription_credits(p_student_id UUID, p_credits INTEGER)
RETURNS void AS $$
BEGIN
  UPDATE student_credits SET total_credits = p_credits, used_credits = 0, reserved_credits = 0, updated_at = now()
  WHERE student_id = p_student_id;
  IF NOT FOUND THEN RAISE EXCEPTION 'student_credits_not_found'; END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER SET search_path = public;
```

All DDL uses `IF NOT EXISTS` / `CREATE OR REPLACE` for safe re-runs.
