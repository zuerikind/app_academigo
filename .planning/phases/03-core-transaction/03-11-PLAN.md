Server Component:
```typescript
// Auth guard
const profile = await requireRoleFromParams("teacher", raw);
const supabase = await createClient();
// Fetch teacher record
const teacher = await supabase.from("teachers").select("id").eq("profile_id", profile.id).maybeSingle();
// Fetch earnings and pending balance in parallel
const [earnings, pendingBalance] = await Promise.all([
  getTeacherEarnings(teacher.id),
  getTeacherPendingBalance(teacher.id),
]);
// Check for existing pending payout request
const { data: pendingPayout } = await supabase
  .from("payout_requests")
  .select("id, status, created_at")
  .eq("teacher_id", teacher.id)
  .eq("status", "pending")
  .maybeSingle();
```
