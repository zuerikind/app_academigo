**Step 1 — lib/queries/admin.ts:**
Read the full file. Add `getMissingMeetLinks` function at the end:

```typescript
export async function getMissingMeetLinks() {
  const supabase = await createAdminClient();
  const now = new Date().toISOString();
  const { data, error } = await supabase
    .from("bookings")
    .select(`
      id, start_time,
      teachers ( id, profiles ( full_name ) ),
      students ( profiles ( full_name ) )
    `)
    .eq("status", "confirmed")
    .is("meeting_link", null)
    .gte("start_time", now)
    .order("start_time", { ascending: true });
  if (error) throw error;
  return data ?? [];
}
```
